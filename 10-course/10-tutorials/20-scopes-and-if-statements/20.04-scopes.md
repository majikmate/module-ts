# Scopes

```typescript
// Outer scope
const age = 20;
const hasLicense = true;

if (age >= 18) {
    // Block scope inside if-statement
    const message = "You are an adult";
    console.log(message); // ✅ Works - message is defined in this block
    console.log(age); // ✅ Works - age is accessible from outer scope
}

// console.log(message);  // ❌ Error! message only exists inside the if-block

if (hasLicense) {
    const status = "You can drive";
    console.log(status); // ✅ Works
} else {
    const status = "You cannot drive"; // Different variable in else-block
    console.log(status); // ✅ Works
}

// console.log(status);   // ❌ Error! status doesn't exist outside the blocks

// Example: Variable Shadowing
const userName = "Alice"; // Outer scope
console.log(userName); // Prints: Alice

if (true) {
    const userName = "Bob"; // Block scope - shadows outer variable
    console.log(userName); // Prints: Bob (uses the inner variable)
}

console.log(userName); // Prints: Alice (outer variable is unchanged)
```
