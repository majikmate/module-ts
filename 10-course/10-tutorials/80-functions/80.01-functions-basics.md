# Functions Basics

```typescript
// Define a function that greets a person and
// receives a name of type string as parameter and
// has no return parameter
function greet(name: string) {
    console.log(`Hello, ${name}!`);
}

// Call the function
greet("Alice");

// Define a function that
// receives to values of type number (a, b) and
// returns a value of type number
function add(a: number, b: number): number {
    return a + b;
}

// Use the returned value
const sum = add(5, 3);
console.log(`Sum: ${sum}`);
```
